from flask import Flask, request, jsonify, render_template_string from datetime import datetime import random import re app = Flask(__name__) # --------------------------------------------------------- # Cloud-Based Cybersecurity Threat Detection System # --------------------------------------------------------- security_logs = [] threat_statistics = { "total_events": 0, "safe": 0, "suspicious": 0, "critical": 0 } # Suspicious IP addresses for demonstration BLACKLISTED_IPS = { "192.168.100.50", "10.10.10.99", "172.16.0.99" } # Suspicious keywords SUSPICIOUS_KEYWORDS = [ "malware", "ransomware", "phishing", "sql injection", "brute force", "port scan", "ddos", "unauthorized", "exploit", "virus" ] def detect_threat(ip, activity): """ Detects threats using simple rule-based analysis. """ activity_lower = activity.lower() score = 0 reasons = [] # IP detection if ip in BLACKLISTED_IPS: score += 50 reasons.append("Blacklisted IP address") # Keyword detection for keyword in SUSPICIOUS_KEYWORDS: if keyword in activity_lower: score += 30 reasons.append(f"Suspicious activity: {keyword}") # SQL injection patterns sql_patterns = [ "' or '1'='1", "union select", "drop table", "select * from", "--" ] for pattern in sql_patterns: if pattern in activity_lower: score += 40 reasons.append("Possible SQL injection") # Brute force detection if "failed login" in activity_lower: score += 25 reasons.append("Repeated failed login attempt") # Random network anomaly simulation if "port scan" in activity_lower: score += 40 reasons.append("Network port scanning detected") # Determine threat level if score >= 70: level = "Critical" threat_statistics["critical"] += 1 elif score >= 30: level = "Suspicious" threat_statistics["suspicious"] += 1 else: level = "Safe" threat_statistics["safe"] += 1 threat_statistics["total_events"] += 1 if not reasons: reasons.append("No known threat detected") return level, score, reasons def add_log(ip, activity): level, score, reasons = detect_threat(ip, activity) log = { "time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"), "ip": ip, "activity": activity, "level": level, "score": score, "reasons": reasons } security_logs.insert(0, log) # Keep latest 50 logs if len(security_logs) > 50: security_logs.pop() return log # --------------------------------------------------------- # Dashboard HTML # --------------------------------------------------------- HTML = """ Cloud Cybersecurity Threat Detection

Real-Time Security Monitoring & Threat Analysis System

Total Security Events

{{ stats.total_events }}

{{ stats.safe }}

🔍 Threat Detection Scanner

🛡 Security Event Logs

{% for log in logs %} {% endfor %}
Time IP Address Activity Threat Level Score
{{ log.time }} {{ log.ip }} {{ log.activity }} {% if log.level == "Safe" %} SAFE {% elif log.level == "Suspicious" %} SUSPICIOUS {% else %} CRITICAL {% endif %} {{ log.score }}
Cloud-Based Cybersecurity Threat Detection System © 2026
""" # --------------------------------------------------------- # Routes # --------------------------------------------------------- @app.route("/") def home(): return render_template_string( HTML, stats=threat_statistics, logs=security_logs ) @app.route("/scan", methods=["POST"]) def scan(): data = request.get_json() ip = data.get("ip", "") activity = data.get("activity", "") log = add_log(ip, activity) return jsonify(log) @app.route("/data") def data(): return jsonify({ "stats": threat_statistics, "logs": security_logs }) # --------------------------------------------------------- # Demo security events # --------------------------------------------------------- add_log( "192.168.1.10", "Normal user login" ) add_log( "192.168.100.50", "Unauthorized login attempt" ) add_log( "10.10.10.99", "Port scan detected" ) # --------------------------------------------------------- # Start Server # --------------------------------------------------------- if __name__ == "__main__": print("--------------------------------------------") print(" Cloud-Based Cybersecurity Threat Detection") print("--------------------------------------------") print("Server running at:") print("http://127.0.0.1:5000") print("--------------------------------------------") app.run( host="127.0.0.1", port=5000, debug=True )
